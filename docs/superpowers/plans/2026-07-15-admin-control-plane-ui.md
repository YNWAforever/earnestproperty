# Admin Control Plane UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a permission-aware `/admin/operations` interface for health monitoring, durable job recovery, sanitized audit review, and two-step registered migration execution.

**Architecture:** Keep one URL-backed Operations route and split its four tabs into focused components. Extend the existing control-plane read responses with safe capabilities, job summaries, and migration states; use one visibility-aware pulse in the route, a typed same-origin API client, and panel-local mutation/filter state.

**Tech Stack:** TypeScript 5.8, React 19, TanStack Start/Router 1.167+, Radix/shadcn UI, Lucide React, Node 24 test runner, Bun test, Neon Postgres, Vite 7.

## Global Constraints

- Use one sidebar entry and one route: `/admin/operations`.
- URL tabs are exactly `overview`, `jobs`, `audit`, and `migrations`.
- Agent sees Overview; Manager sees Overview, Jobs, Audit; Admin sees all four tabs.
- Hidden sections must not call endpoints the current role cannot access.
- Backend permission checks remain authoritative and default deny.
- Poll every 30 seconds only while `document.visibilityState === "visible"`.
- Support single-job retry and cancel only; do not add bulk actions.
- Migration approval tokens stay in component memory only and are cleared before Apply awaits.
- Never render job payloads, raw provider errors, SQL, prompts, phone numbers, secrets, or complete customer records.
- Do not apply `20260714180000_backend_control_plane.sql` to production in this plan.
- Do not run mutating integration tests without an explicit `TEST_DATABASE_URL`.
- Follow TDD: observe each new test fail for the missing behavior before production edits.
- Preserve unrelated dirty SDD files and generated noise.

```ts
export async function getJobSummary() {
  const { queryRows } = await import("../neon/db.server.ts");
  const [countRows, attentionRows] = await Promise.all([
    queryRows(`SELECT
      count(*) FILTER (WHERE status = 'queued')::int AS queued,
      count(*) FILTER (WHERE status = 'running')::int AS running,
      count(*) FILTER (WHERE status = 'succeeded')::int AS succeeded,
      count(*) FILTER (WHERE status = 'failed')::int AS failed,
      count(*) FILTER (WHERE status = 'cancelled')::int AS cancelled
      FROM ops_jobs`),
    queryRows(`SELECT id::text AS id, job_type, status, attempt_count, max_attempts,
        lease_expires_at, last_error_code, updated_at
      FROM ops_jobs
      WHERE status = 'failed'
         OR (status = 'running' AND lease_expires_at < now())
      ORDER BY updated_at DESC, id DESC
      LIMIT 5`),
  ]);
  const counts = countRows[0] ?? {};
  return {
    counts: {
      queued: Number(counts.queued ?? 0),
      running: Number(counts.running ?? 0),
      succeeded: Number(counts.succeeded ?? 0),
      failed: Number(counts.failed ?? 0),
      cancelled: Number(counts.cancelled ?? 0),
    },
    attention: attentionRows.map((row) => ({
      id: String(row.id),
      jobType: String(row.job_type),
      status: String(row.status),
      attemptCount: Number(row.attempt_count),
      maxAttempts: Number(row.max_attempts),
      leaseExpiresAt: row.lease_expires_at == null ? null : isoDate(row.lease_expires_at),
      errorCode: row.last_error_code == null ? null : String(row.last_error_code),
      updatedAt: isoDate(row.updated_at),
    })),
  };
}
```

```ts
export type MigrationState = {
  id: string;
  checksum: string;
  dependencies: string[];
  summary: string;
  postconditions: RegisteredMigration["postconditions"];
  status: "pending" | "applied" | "drift";
};

export async function listMigrationStates(
  deps: Pick<MigrationServiceDependencies, "registry" | "queryRows"> = {},
): Promise<MigrationState[]> {
  const resolved = await resolveDependencies(deps);
  const [appliedRows, runRows] = await Promise.all([
    resolved.queryRows<{ version: unknown }>("SELECT version FROM app_migrations"),
    resolved.queryRows<{ migration_id: unknown; checksum: unknown }>(
      "SELECT DISTINCT ON (migration_id) migration_id, checksum FROM ops_migration_runs WHERE result = 'succeeded' ORDER BY migration_id, finished_at DESC NULLS LAST, id DESC",
    ),
  ]);
  const applied = new Set(appliedRows.map((row) => String(row.version)));
  const runChecksums = new Map(
    runRows.map((row) => [String(row.migration_id), String(row.checksum)]),
  );
  return resolved.registry.map((migration) => ({
    id: migration.id,
    checksum: migration.checksum,
    dependencies: [...migration.dependencies],
    summary: migration.summary,
    postconditions: migration.postconditions.map((item) => ({ ...item })),
    status:
      runChecksums.has(migration.id) && runChecksums.get(migration.id) !== migration.checksum
        ? "drift"
        : applied.has(migration.id) || runChecksums.has(migration.id)
          ? "applied"
          : "pending",
  }));
}
```

```ts
import type { OperationsCapabilities } from "../../control-plane/capabilities.ts";
import type { OperationTab } from "./operations-types.ts";

export function allowedOperationTabs(capabilities: OperationsCapabilities): OperationTab[] {
  const tabs: OperationTab[] = ["overview"];
  if (capabilities.jobsRead) tabs.push("jobs");
  if (capabilities.auditRead) tabs.push("audit");
  if (capabilities.migrationsPlan) tabs.push("migrations");
  return tabs;
}

export function resolveOperationTab(
  value: unknown,
  capabilities: OperationsCapabilities,
): OperationTab {
  const allowed = allowedOperationTabs(capabilities);
  return typeof value === "string" && allowed.includes(value as OperationTab)
    ? (value as OperationTab)
    : allowed[0];
}
```

Create `operations-client.ts` with `credentials: "same-origin"`, JSON headers, envelope validation, and these exports:

```ts
export class OperationsClientError extends Error {
  constructor(
    message: string,
    readonly status: number,
    readonly code: string,
    readonly requestId: string | null,
    readonly retryable: boolean,
  ) {
    super(message);
    this.name = "OperationsClientError";
  }
}

export async function requestControlPlane<T>(
  path: string,
  init: RequestInit = {},
  fetchImpl: typeof fetch = fetch,
): Promise<{ data: T; requestId: string }> {
  const response = await fetchImpl(`/api/admin/control-plane${path}`, {
    credentials: "same-origin",
    ...init,
    headers: {
      accept: "application/json",
      ...(init.body ? { "content-type": "application/json" } : {}),
      ...Object.fromEntries(new Headers(init.headers).entries()),
    },
  });
  const body = (await response.json().catch(() => null)) as Record<string, unknown> | null;
  const requestId = typeof body?.requestId === "string" ? body.requestId : null;
  if (!body || typeof body.ok !== "boolean") {
    throw new OperationsClientError("未能讀取系統回應。", response.status, "INVALID_RESPONSE", requestId, false);
  }
  if (!body.ok) {
    const error = body.error && typeof body.error === "object" ? body.error as Record<string, unknown> : {};
    throw new OperationsClientError(
      typeof error.message === "string" ? error.message : "操作未能完成。",
      response.status,
      typeof error.code === "string" ? error.code : "INTERNAL_ERROR",
      requestId,
      error.retryable === true,
    );
  }
  if (!requestId || !("data" in body)) {
    throw new OperationsClientError("未能讀取系統回應。", response.status, "INVALID_RESPONSE", requestId, false);
  }
  return { data: body.data as T, requestId };
}
```

```ts
export function useOperationsPulse(intervalMs = 30_000) {

  useEffect(() => {
    const poller = createOperationsPoller({
      intervalMs,
      isVisible: () => document.visibilityState === "visible",
      subscribeVisibility: (listener) => {

      },
      setTimer: (listener, ms) => window.setInterval(listener, ms),
      clearTimer: (id) => window.clearInterval(id),
      onPulse: () => setPulse((value) => value + 1),
    });

    return () => poller.stop();
  }, [intervalMs]);
```
